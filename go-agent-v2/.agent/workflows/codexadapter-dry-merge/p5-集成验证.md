```bash
python3 - <<'PY'
import glob, os, sys
base = "internal/apiserver/codexadapter"
files = [os.path.basename(p) for p in glob.glob(base + "/*.go")]

def owner(f: str) -> str:
    if f in ("adapter.go", "adapter_client.go", "runtime_context.go"):
        return "P0"
    if f.startswith("thread_") or f == "thread_archive_utils_guardrail_test.go":
        return "P1"
    if f.startswith("turn_tracker") or f in ("turn_tracker_test.go", "tracked_turn_shape_guardrail_test.go"):
        return "P3"
    if ((f.startswith("turn_") and not f.startswith("turn_tracker")) or
        f.startswith("slash_command") or
        f in ("with_thread.go", "with_thread_test.go", "turn_prompt_test.go", "turn_resume_test.go")):
        return "P2"
    return ""

uncovered = sorted([f for f in files if not owner(f)])
if uncovered:
    print("UNCOVERED FILES:")
    for f in uncovered:
        print(" -", f)
    sys.exit(1)
print("OK: uncovered = 0")
PY
```
