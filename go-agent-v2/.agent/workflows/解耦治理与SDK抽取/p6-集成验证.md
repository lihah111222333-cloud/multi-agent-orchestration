4. 按包统计对比
   ```bash
   for target in internal/tools internal/tooladapter internal/difftracker internal/lsp internal/apiserver internal/apiserver/codexadapter internal/codex internal/uistate; do
     count=$(find "./$target" -name '*.go' ! -name '*_test.go' 2>/dev/null -exec cat {} + | sed '/^[[:space:]]*$/d; /^[[:space:]]*\/\//d; /^[[:space:]]*\/\*/d; /^[[:space:]]*\*/d' | wc -l | tr -d ' ')

   done | sort -rn
   ```

5. 依赖验证
   ```bash
   # tools 不应 import internal/executor, runner, service, store
   if rg -n '"github.com/multi-agent/go-agent-v2/internal/(executor|runner|service|store)"' internal/tools/*.go; then
     echo "FAIL tools dependency boundary"
   else
     echo "PASS tools dependency boundary"
   fi

   # difftracker 不应 import 任何 internal/*
   if rg -n '"github.com/multi-agent/go-agent-v2/internal/' internal/difftracker/*.go; then
     echo "FAIL difftracker dependency boundary"
   else
     echo "PASS difftracker dependency boundary"
   fi

   # lsp 不应 import 任何 internal/*
   # 注意: lsp 当前已是零 internal 依赖。若此检查失败，说明 P2 合并时引入了回归。
   if rg -n '"github.com/multi-agent/go-agent-v2/internal/' internal/lsp/*.go; then
     echo "FAIL lsp dependency boundary (P2 regression!)"
   else
     echo "PASS lsp dependency boundary"
   fi

   # bus 不应 import internal/codex
   if rg -n '"github.com/multi-agent/go-agent-v2/internal/codex"' internal/bus/*.go; then
     echo "FAIL bus dependency boundary (still imports codex)"
   else
     echo "PASS bus dependency boundary"
   fi

   # codex 不应反向依赖 internal/bus（避免重建反向耦合）
   if rg -n '"github.com/multi-agent/go-agent-v2/internal/bus"' internal/codex/*.go; then
     echo "FAIL codex reverse dependency boundary"
   else
     echo "PASS codex reverse dependency boundary"
   fi
   ```
